Rest Parameters

lets see if we can handle infinite parameters

 Rest parameters let a function accept any number of arguments.
  They collect all extra arguments into an array.




     Syntax: ...paramName (three dots before the name)
  The rest parameter must always be the LAST parameter.
console.log("\n=== REST PARAMETERS ===");

function sum(...numbers) {
  return numbers.reduce((total, n) => total + n, 0);
}

console.log("sum(1,2,3):", sum(1, 2, 3));           // 6
console.log("sum(1,2,3,4,5):", sum(1, 2, 3, 4, 5)); // 15
console.log("sum(10,20):", sum(10, 20));             // 30

// Mix of named and rest parameters
function introduce(greeting, ...names) {
  return `${greeting}, ${names.join(" and ")}!`;
}

console.log(introduce("Hello", "Ayesha"));               // Hello, Ayesha!
console.log(introduce("Hi", "Ayesha", "Sara", "Omar"));  // Hi, Ayesha and Sara and Omar!
